function doGet(e) { return ContentService .createTextOutput(JSON.stringify({ status: "ok" })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const body = JSON.parse(e.postData.contents || "{}"); const userText = String(body.text || "").trim(); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Behavior"); const data = sheet.getDataRange().getValues(); let reply = "……"; let exitType = "none"; let sessionUrl = ""; for (let i = 1; i < data.length; i++) { const trigger = String(data[i][0] || ""); const r = String(data[i][2] || ""); const q = String(data[i][3] || ""); const e = String(data[i][4] || ""); const url = String(data[i][5] || ""); if (!trigger) continue; if (userText.includes(trigger)) { reply = r + (q ? "\n\n" + q : ""); exitType = e || "none"; sessionUrl = url || ""; break; } } // fallback if (!reply || reply === "……") { reply = "……もう少し聞いてもいいですか。"; } return ContentService .createTextOutput(JSON.stringify({ reply: reply, exit_type: exitType, session_url: sessionUrl })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ reply: "ひいろです。一緒にどんなこと、お話しますか？", exit_type: "none", session_url: "" })) .setMimeType(ContentService.MimeType.JSON); } }
