async function send() { const input = document.getElementById('msg'); const text = input.value.trim(); if (!text) return; addMessage(text, 'user'); input.value = ''; try { const res = await fetch(GAS_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ text: text, sessionId: "session_" + Date.now() }) }); // ■ ここが重要（生レス確認） const raw = await res.text(); console.log("RAW RESPONSE:", raw); let data = {}; try { data = JSON.parse(raw); } catch (e) { console.log("JSON PARSE FAILED"); } console.log("PARSED:", data); addMessage(data.reply || "……", "system"); } catch (err) { console.log("FETCH ERROR:", err); addMessage("……", "system"); } }
